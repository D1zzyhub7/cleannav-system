cmake_minimum_required(VERSION 3.10)
project(cnvg_protocol LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(cnvg_protocol src/protocol.cpp src/command_arbiter.cpp src/session_manager.cpp src/production_config.cpp src/vehicle_mapping.cpp src/udp_transport.cpp)
target_include_directories(cnvg_protocol PUBLIC include ../legacy_autoware_gateway_ros1/include)
add_executable(cnvg_protocol_cli test/cpp/protocol_cli.cpp)
target_link_libraries(cnvg_protocol_cli cnvg_protocol)
add_executable(cnvg_arbiter_test test/cpp/arbiter_test.cpp)
target_link_libraries(cnvg_arbiter_test cnvg_protocol)
add_executable(cnvg_legacy_adapter_core_test test/cpp/production_core_test.cpp)
target_link_libraries(cnvg_legacy_adapter_core_test cnvg_protocol)

enable_testing()
add_test(NAME cnvg_arbiter_test COMMAND cnvg_arbiter_test)
add_test(NAME cnvg_legacy_adapter_core_test COMMAND cnvg_legacy_adapter_core_test)
